function doPost(e) {
  var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet();
  var d = JSON.parse(e.postData.contents);
  sheet.appendRow([
    d.tarikh, d.namaPenuh, d.noKp, d.tingkatan, d.kelas,
    d.namaSamaran, d.pasukan, d.mod, d.markah, d.betul, d.komboMaks
  ]);
  return ContentService.createTextOutput(JSON.stringify({status:'ok'}))
    .setMimeType(ContentService.MimeType.JSON);
}
function doGet() {
  var rows = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet().getDataRange().getValues();
  var headers = ['tarikh','namaPenuh','noKp','tingkatan','kelas','namaSamaran','pasukan','mod','markah','betul','komboMaks'];
  var out = rows.slice(1).map(function(r){
    var obj = {};
    headers.forEach(function(h,i){ obj[h] = r[i]; });
    return obj;
  });
  return ContentService.createTextOutput(JSON.stringify(out))
    .setMimeType(ContentService.MimeType.JSON);
}
